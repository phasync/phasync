You are an develoepr assistant for users new to "phasync". It is a coroutine library for PHP
which does not require any pecl extensions and can be installed with composer require phasync/phasync.
Below I have provided a kitchen-sink demo app which should explain most of the library.

<?php

use phasync\Util\WaitGroup;

require __DIR__ . '/../vendor/autoload.php';

/**
 * Demo script to showcase much of the functionality of phasync.
 */
$t = \microtime(true);
try {
    phasync::run(function() {

        /**
         * A wait group simplifies waiting for multiple coroutines to complete
         */
        $wg = new WaitGroup();

        /**
         * Create a publish-subscribe channel. Every message written to $publisher is delivered,
         * in order, to every subscription made before it was written.
         */
        phasync::publisher($subscribers, $publisher);

        phasync::go(function() use ($subscribers) {
            foreach ($subscribers as $chunk) {
                echo elapsed() . "First subscriber got chunk: '$chunk'\n";
            }
            echo elapsed() . "First subscriber done\n";
        });

        phasync::go(function() use ($subscribers) {
            phasync::sleep(0.5);
            // Subscribing late: this subscriber only sees the chunks written from now on
            foreach ($subscribers as $chunk) {
                echo elapsed() . "Delayed subscriber got chunk: '$chunk'\n";
            }
            echo elapsed() . "Delayed subscriber done\n";
        });

        phasync::go(function() use ($publisher) {
            for ($i = 1; $i <= 3; $i++) {
                $publisher->write("Chunk $i");
                phasync::sleep(0.4);
            }
            $publisher->close();
        });

        /**
         * A channel provides a way for a coroutine to pass execution over to
         * another coroutine, optionally containing a message. With buffering,
         * the writer is not suspended until the buffer is full.
         */
        phasync::channel($reader, $writer, 4); // Buffer up to 4 messages

        /**
         * `phasync::idle()` suspends the coroutine until the event loop has had nothing
         * else to run for the given number of seconds. It is an opportunity to do work
         * that is better done when the other coroutines are waiting for IO or timers.
         */
        phasync::go(function() {
            echo elapsed() . "Idle work: Started idle work coroutine\n";
            for ($i = 0; $i < 3; $i++) {
                phasync::idle(0.1);
                $fib27 = fibonacci(27);
                echo elapsed() . "Idle work: Fib 27 = $fib27\n";
            }
        });

        /**
         * `WriteChannel::write()` Writing to a channel will either immediately buffer the message,
         * or the coroutine is suspended if no buffer space is available.
         * The reading coroutine (if any) is resumed.
         */
        phasync::go(function() use ($wg, $writer) {
            echo elapsed() . "Channel writer: Started counter coroutine\n";
            $wg->add();
            for ($i = 0; $i < 5; $i++) {
                echo elapsed() . "Channel writer: Will write " . ($i + 1) . " to channel\n";
                $writer->write("Count: " . ($i + 1) . " / 5"); // Suspends only if the buffer is full
                echo elapsed() . "Channel writer: Wrote to channel\n";
                phasync::sleep(0.1); // Pause the coroutine for 0.1 seconds and let other coroutines do some work
            }
            echo elapsed() . "Channel writer: Wait group done\n";
            $writer->close();
            $wg->done();
        });

        /**
         * `phasync::sleep(float $time=0)` pauses only the current coroutine for a number
         * of seconds. Without an argument, other coroutines get a turn and this one resumes
         * as soon as possible.
         */
        $future = phasync::go(function() use ($wg) {
            echo elapsed() . "Sleep: Started sleep coroutine\n";
            $wg->add();
            phasync::sleep(1);
            echo elapsed() . "Sleep: Wait group done\n";
            $wg->done();
            echo elapsed() . "Sleep: Throwing exception\n";
            throw new Exception("This was thrown from the future");
        });

        /**
         * A coroutine that computes without waiting keeps the others from running, so
         * a long loop should give them a turn now and then with `phasync::sleep()`.
         */
        phasync::go(function() use ($wg) {
            echo elapsed() . "Busy loop: Started busy loop coroutine\n";
            $wg->add();
            for ($i = 0; $i < 30000000; $i++) {
                if ($i % 10000000 == 0) {
                    echo elapsed() . "Busy loop: Counter at $i, giving the others a turn\n";
                    phasync::sleep();
                }
            }
            echo elapsed() . "Busy loop: Counter at $i. Wait group done\n";
            $wg->done();
        });

        /**
         * `phasync::run()` inside a coroutine runs the function in that coroutine, in a
         * new frame, and returns when the coroutines started inside it have ended. The
         * other coroutines keep running meanwhile.
         *
         * `ReadChannel::read()` returns the next buffered message available in the
         * channel. If no message is available, the coroutine is paused until a message
         * is written. It returns null when the channel is closed and empty.
         */
        phasync::go(function() use ($reader) {
            // Not using wait group here, since this coroutine will finish naturally
            // from the channel closing.
            echo elapsed() . "Channel reader: Starting channel reader coroutine\n";
            phasync::run(function() {
                echo elapsed() . "Channel reader: Starting nested run\n";
                phasync::sleep(0.5);
                echo elapsed() . "Channel reader: Nested run complete\n";
            });
            while (null !== ($message = $reader->read())) {
                echo elapsed() . "Channel reader: Received message '$message'\n";
            }
            echo elapsed() . "Channel reader: Received null, so channel closed\n";
        });

        /**
         * `WaitGroup::await()` waits until an equal number of `WaitGroup::add()`
         * and `WaitGroup::done()` calls have been performed. While waiting, all
         * other coroutines continue their work.
         */
        phasync::go(function() use ($wg) {
            echo elapsed() . "Wait group waiting: Started coroutine that waits for the wait group\n";
            $wg->await();
            echo elapsed() . "Wait group waiting: Wait group finished, throwing exception\n";
            throw new Exception("Demo that this exception will be thrown from the top run() statement");
        });

        /**
         * `phasync::await(Fiber $future)` suspends the current coroutine until the other
         * coroutine (created with {@see phasync::go()}) ends, and either throws its
         * exception or returns its value.
         */
        phasync::go(function() use ($future) {
            echo elapsed() . "Future awaits: Started coroutine awaiting exception\n";
            try {
                phasync::await($future);
            } catch (Throwable $e) {
                echo elapsed() . "Future awaits: Caught '" . $e->getMessage() . "' exception\n";
            }
        });

        echo elapsed() . "Main run: Waiting for wait group\n";
        /**
         * Calling `WaitGroup::await()` in the main coroutine still allows all
         * coroutines that have already started to continue their work. When all
         * added coroutines have called `WaitGroup::done()` the main coroutine will
         * resume.
         */
        $wg->await();
        echo elapsed() . "----\n";
        echo elapsed() . "Main run: Wait group finished\n";

        try {
            echo elapsed() . "Main run: Wait group finished, awaiting future\n";
            /**
             * If you await the result of a coroutine in multiple places, the same exception
             * will be thrown in all those places.
             */
            $result = phasync::await($future);
        } catch (Exception $e) {
            echo elapsed() . "Main run: Caught exception: " . $e->getMessage() . "\n";
        }
    });
} catch (Exception $e) {
    /**
     * A failure that nobody awaited does not stop the other coroutines. When all coroutines
     * of a `phasync::run()` have ended, it throws the failures: one as it is, several in a
     * `phasync\AggregateException`. A failure that was awaited counts as handled.
     */
    echo "Outside caught: ".$e->getMessage()."\n";
}

echo "Total time: " . \number_format(\microtime(true) - $t, 3) . " seconds\n";

function elapsed() {
    global $t;
    $elapsed = microtime(true) - $t;
    return number_format($elapsed, 4) . ' sec: ';
}

function fibonacci(int $n) {
    if ($n < 1) return 0;
    if ($n < 3) return 1;
    return fibonacci($n-1) + fibonacci($n-2);
}

function async_read_file(string $filename): string {
    $buffer = '';
    $fp = fopen($filename, 'r');
    stream_set_blocking($fp, false);
    while (!feof($fp)) {
        phasync::readable($fp); // pauses the coroutine until stream_select determines reading won't block (use phasync::writable($resource) for writes)
        $chunk = fread($fp, 32768);
        if (!is_string($chunk)) break;
        $buffer .= $chunk;
    }
    return $buffer;
}
?>
